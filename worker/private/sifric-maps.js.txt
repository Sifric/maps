import { registerPlaceImages, stylePlaces } from "/place-markers.js";

const libraryLoads = new Map();

function loadScript(url) {
  if (!libraryLoads.has(url)) {
    const pending = new Promise((resolve, reject) => {
      const script = document.createElement("script");
      script.src = url;
      script.onload = resolve;
      script.onerror = () => { script.remove(); reject(new Error(`Could not load ${url}`)); };
      document.head.append(script);
    }).catch(error => { libraryLoads.delete(url); throw error; });
    libraryLoads.set(url, pending);
  }
  return libraryLoads.get(url);
}

const registered = new WeakSet();

export async function createSifricMap(container, options = {}) {
  const base = new URL(options.baseUrl ?? import.meta.url);
  const css = new URL("/vendor/maplibre-gl.css", base).href;
  if (![...document.querySelectorAll('link[rel="stylesheet"]')].some(link => link.href === css)) {
    const link = document.createElement("link");
    link.rel = "stylesheet";
    link.href = css;
    document.head.append(link);
  }
  const [, , styleResponse] = await Promise.all([
    globalThis.maplibregl ? undefined : loadScript(new URL("/vendor/maplibre-gl.js", base).href),
    globalThis.pmtiles ? undefined : loadScript(new URL("/vendor/pmtiles.js", base).href),
    fetch(new URL("/style.json", base)),
  ]);
  const library = globalThis.maplibregl;
  if (!registered.has(library)) {
    library.addProtocol("pmtiles", new globalThis.pmtiles.Protocol().tile);
    registered.add(library);
  }
  const { baseUrl, ...mapOptions } = options;
  if (!styleResponse.ok) throw new Error("Could not load the Sifric map style.");
  const style = stylePlaces(await styleResponse.json());
  const map = new library.Map({
    center: [80.7, 7.9],
    zoom: 7,
    maxZoom: 20,
    renderWorldCopies: false,
    ...mapOptions,
    container,
    style,
  });
  registerPlaceImages(map);
  return map;
}

export function createSifricClient(apiBase) {
  const base = new URL(apiBase, location.href);
  async function request(path, query = {}, init = {}) {
    const url = new URL(`/developer/api/v1/${path}`, base);
    url.search = new URLSearchParams(Object.entries(query).filter(([, value]) => value != null));
    const response = await fetch(url, init);
    const result = await response.json();
    if (!response.ok) throw new Error(result.error || `Sifric API returned ${response.status}`);
    return result;
  }

  function post(path, data, signal) {
    return request(path, {}, {
      method: "POST", headers: { "Content-Type": "application/json" },
      body: JSON.stringify(data), signal,
    });
  }

  return {
    search: (q, { limit, signal } = {}) => request("places/search", { q, limit }, { signal }),
    nearby: (lon, lat, { radius, limit, signal } = {}) =>
      request("places/nearby", { lon, lat, radius, limit }, { signal }),
    details: (id, signal) => request(`places/${encodeURIComponent(id)}`, {}, { signal }),
    photos: (id, signal) => request(`places/${encodeURIComponent(id)}/photos`, {}, { signal }),
    route: (data, signal) => post("routes", data, signal),
    matrix: (data, signal) => post("matrix", data, signal),
    matchTrace: (data, signal) => post("roads/match", data, signal),
    optimizeFleet: (data, signal) => post("optimize", data, signal),
    validateAddress: (data, signal) => post("address/validate", data, signal),
    locateByCell: (data, signal) => post("geolocate", data, signal),
  };
}
