import { createMapOptions } from "/map-options.js";
import { selectedPlaceElement } from "/place-markers.js";
import { createSifricMap } from "./sifric-maps.js";

const query = new URLSearchParams(location.search);
const read = (name, min, max) => {
  const raw = query.get(name);
  const value = Number(raw);
  return raw?.trim() && Number.isFinite(value) && value >= min && value <= max ? value : undefined;
};
const lon = read("lon", -180, 180);
const lat = read("lat", -85, 85);
const zoom = read("zoom", 0, 20);
const center = lon !== undefined && lat !== undefined ? [lon, lat] : undefined;
const staticImage = query.get("static") === "1";
const attributionPreference = query.get("attribution");

const status = document.getElementById("status");
const view = document.getElementById("map-view");
const theme = document.getElementById("map-theme");
function showStatus(message) {
  status.hidden = false;
  status.textContent = message;
}

try {
  const map = await createSifricMap("map", {
    attributionControl: false,
    ...(center ? { center } : {}),
    ...(zoom !== undefined ? { zoom } : {}),
  });
  const wordmark = document.createElement("span");
  wordmark.id = "sifric-wordmark";
  wordmark.textContent = "Sifric";
  map.getContainer().append(wordmark);

  let attribution;
  let attributionMode;
  function showAttribution(type) {
    const mode = staticImage || type === "satellite" || attributionPreference === "1" ? "full" :
      attributionPreference === "0" ? "collapse" : "auto";
    if (mode === attributionMode) return;
    attributionMode = mode;
    if (attribution) map.removeControl(attribution);
    const control = new maplibregl.AttributionControl({
      ...(mode !== "auto" && { compact: mode === "collapse" }),
      customAttribution: '<a href="https://www.openstreetmap.org/copyright" target="_blank" rel="noopener">© OpenStreetMap contributors</a>',
    });
    attribution = control;
    map.addControl(control);
    if (mode === "collapse") {
      map.once("idle", () => setTimeout(() => {
        if (attribution !== control) return;
        map.getContainer().querySelector(".maplibregl-compact-show > summary")?.click();
      }, 5000));
    }
  }
  showAttribution(query.get("type"));
  map.once("load", () => {
    const appearance = createMapOptions(map, {
      baseStyle: map.getStyle(),
      status: showStatus,
      embed: {
        theme: query.get("theme"),
        type: query.get("type"),
        onChange(settings) {
          showAttribution(settings.type);
          view.value = settings.type;
          theme.value = settings.theme;
          for (const option of view.options) option.disabled = settings.unavailable.includes(option.value);
        },
      },
    });
    map.on("error", event => appearance.handleError(event));
    view.addEventListener("change", () => appearance.setType(view.value));
    theme.addEventListener("change", () => appearance.setTheme(theme.value));
    document.getElementById("appearance").hidden = query.get("controls") !== "1";
    map.once("idle", () => { document.getElementById("map").dataset.ready = "true"; });
  });
  if (!staticImage && query.get("zoomControls") === "1") {
    map.addControl(new maplibregl.NavigationControl({ showCompass: false }));
  }
  if (center && query.get("pin") === "1") {
    new maplibregl.Marker({ element: selectedPlaceElement(), anchor: "bottom" })
      .setLngLat(center).addTo(map);
  }
} catch {
  showStatus("The map could not be loaded.");
}
